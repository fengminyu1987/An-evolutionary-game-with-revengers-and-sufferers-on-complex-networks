import matplotlib.pyplot as plt
import numpy as np

np.random.seed(123456789)
data = np.random.rand(100).reshape(10, 10)
print(data)
plt.imshow(data, origin='lower')
plt.xticks(np.arange(0, 10, 1)[::3], np.around(np.arange(0, 1, 0.1)[::3], decimals=1))
plt.yticks(np.arange(0, 10, 1)[::3], np.around(np.arange(0, 1, 0.1)[::3], decimals=1))
plt.colorbar()
plt.show()
